// 🛰️ Singapore Live Telemetry Engine async function getLiveData() { try { // 1. Ask data.gov.sg for the newest air report const response = await fetch('https://api-open.data.gov.sg/v1/environment/psi'); const data = await response.json(); if (data && data.items && data.items.length > 0) { const readings = data.items[0].readings; const psi24 = readings.psi_twenty_four_hourly; const pm25 = readings.pm25_one_hourly; // Pick numbers (Central Singapore by default) const currentPsi = psi24.central || 45; const currentPm25 = pm25.west || pm25.central || 15; // Put the real numbers on your screen! const psiBox = document.getElementById('kpi-psi-value'); if (psiBox) psiBox.textContent = currentPsi; const pm25Box = document.getElementById('kpi-pm25-value'); if (pm25Box) pm25Box.textContent = currentPm25; // Change badge color (Green if clean, Orange if hazy) const badgeText = document.getElementById('kpi-psi-badge-text'); if (badgeText) { badgeText.textContent = currentPsi <= 50 ? 'Good (S.G.)' : (currentPsi <= 100 ? 'Moderate (S.G.)' : 'Unhealthy (S.G.)'); } // Update the station dots on the map if (document.getElementById('label-station-jurong')) { document.getElementById('label-station-jurong').textContent = `Jurong West: ${psi24.west || 45} PSI (${pm25.west || 15} µg/m³)`; } if (document.getElementById('label-station-bishan')) { document.getElementById('label-station-bishan').textContent = `Bishan HQ: ${psi24.central || 45} PSI (${pm25.central || 15} µg/m³)`; } if (document.getElementById('label-station-woodlands')) { document.getElementById('label-station-woodlands').textContent = `Woodlands: ${psi24.north || 45} PSI (${pm25.north || 15} µg/m³)`; } if (document.getElementById('label-station-bedok')) { document.getElementById('label-station-bedok').textContent = `Bedok: ${psi24.east || 45} PSI (${pm25.east || 15} µg/m³)`; } } // 2. Fetch live wind from government weather stations const windRes = await fetch('https://api-open.data.gov.sg/v1/environment/wind-speed'); const windData = await windRes.json(); if (windData?.items?.[0]?.readings?.[0]) { const knots = windData.items[0].readings[0].value; const kmh = Math.round(knots * 1.852); const windBox = document.getElementById('kpi-wind-speed'); if (windBox) windBox.textContent = kmh; } console.log('✅ Real Singapore data loaded successfully!'); } catch (error) { console.log('Trouble fetching live data:', error); } } // Run automatically as soon as the page loads! window.addEventListener('DOMContentLoaded', getLiveData); getLiveData();
